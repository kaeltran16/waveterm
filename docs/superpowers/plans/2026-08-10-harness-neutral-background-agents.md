# Harness-Neutral Background Agents Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace hidden Claude selection with one visible shared harness preference for Runs, Pet Errand, and bare Ask, while removing Sharpen and preserving legacy Claude Runs.

**Architecture:** A new `pkg/harness` catalog owns harness identity, capabilities, executable lookup, and concurrent probing. Consult remains a one-shot executor, while Jarvis owns persistent Run-worker adapters. The frontend uses one persisted preference and one reusable Autonomy-style picker; each new Run copies the selected runtime and never changes it afterward.

**Tech Stack:** Go, wshrpc code generation, React 19, TypeScript, Jotai, Floating UI, Vitest, CDP scenario harness.

## Global Constraints

- Follow `docs/superpowers/specs/2026-08-10-harness-neutral-headless-design.md`.
- Never default new background work to Claude or the first installed harness.
- Never fall back to another harness after an error.
- Keep OpenRouter inside utility inference; do not expose it as an installed harness.
- Existing `Run.Runtime == ""` means legacy Claude only.
- New Runs require an explicit persisted harness and become immutable after Start.
- Run and Pet Errand share `harness:preferredruntime`.
- Explicit `@ask <runtime>` is one-off and does not mutate the preference.
- Do not hand-edit generated files; run `task generate` after Go wire/config changes.
- Use Tailwind theme tokens, not raw colors, in shipping components.
- Do not commit after individual tasks. Under repository policy, batch implementation into one commit only after explicit user approval.
- Preserve unrelated worktree changes.

func List() []Spec {
	return append([]Spec(nil), specs...)
}

func Lookup(runtime string) (Spec, bool) {
	for _, spec := range specs {

		}
	}
	return Spec{}, false

func ValidateInstalled(runtime string, operation Operation) (Spec, error) {
	spec, ok := Lookup(runtime)
	if !ok {
		return Spec{}, fmt.Errorf("unknown harness %q", runtime)

```go
var versionCommand = func(ctx context.Context, bin string) ([]byte, error) {
	return exec.CommandContext(ctx, bin, "--version").CombinedOutput()
}

func probe(ctx context.Context, spec Spec) ProbeResult {
	if _, err := lookPath(spec.Bin); err != nil {
		return ProbeResult{Spec: spec}
	}
	out, _ := versionCommand(ctx, spec.Bin)
	return ProbeResult{Spec: spec, Installed: true, Version: strings.TrimSpace(string(out))}
}

func ProbeAll(ctx context.Context) []ProbeResult {
	results := make([]ProbeResult, len(specs))
	var wg sync.WaitGroup
	for i, spec := range specs {
		wg.Add(1)
		go func(i int, spec Spec) {
			defer wg.Done()
			results[i] = probe(ctx, spec)
		}(i, spec)
	}
	wg.Wait()
	return results
}
```

export type HarnessOperation = "consult" | "run-worker";

export function supportsOperation(h: HarnessInfo, operation: HarnessOperation): boolean {
    return operation === "consult" ? h.consultcapable : h.runworkercapable;
}

export function harnessPickerItems(harnesses: HarnessInfo[], runtime: string, operation: HarnessOperation) {
    return harnesses.map((h) => {
        const supported = supportsOperation(h, operation);
        return {
            runtime: h.runtime,
            label: h.label,
            selected: h.runtime === runtime,
            selectable: h.installed && supported,
            unavailableReason: !supported ? ("unsupported" as const) : !h.installed ? ("not-installed" as const) : undefined,
            disclosure: operation === "run-worker" ? "Can edit files and run commands without approval." : undefined,
        };
    });
}
